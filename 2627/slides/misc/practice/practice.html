Statistics Skills Practice

This page generates statistics practice questions for you. It is for practising calculations and procedures — choosing the right formula, standardising, using the z table and doing the arithmetic. It is not a substitute for practising conceptual questions, which you get separately.

How it works:

  • choose a seed (any whole number), a difficulty, the number of questions and which topics to include;
  • press Generate practice set;
  • type your answer in the box and press Check answer;
  • if you are stuck, ask for a hint, or open the worked solution.

The questions are generated from the seed: the same seed with the same settings always gives exactly the same set of questions. So you can write down “seed 18492, difficulty 2, question 4” and come back to that question later. Generate as many sets as you like — nothing is saved or sent anywhere.

The R helper functions

The five cells below are not shown on the page. They load the contents of the files in R/ into the webR session that runs in the browser, which is where the questions are actually generated. Each cell takes its code from one file with the file chunk option, so the files stay the single source of truth: edit R/normal.R and the change reaches this page the next time you render.

Generate a practice set

{
  // ------------------------------------------------------------------
  // Quarto Live starts a webR session for this page because of the
  // {webr} cells above, and publishes it as the global `webROjs`.  That
  // happens in the page's OJS prelude, which is defined after this cell,
  // so wait for it instead of assuming it already exists.
  // ------------------------------------------------------------------
  const quartoLive = async () => {
    for (let attempt = 0; attempt < 300; attempt++) {
      if (typeof globalThis.webROjs !== "undefined") return globalThis.webROjs;
      await new Promise((resolve) => setTimeout(resolve, 100));
    }
    throw new Error(
      "Quarto Live did not start a webR session. Check that the extension is " +
        "installed and that the {webr} cells above rendered."
    );
  };

  const webR = await (await quartoLive()).webRPromise;

  // Those cells are evaluated as soon as webR is ready.  If a button is
  // pressed before that has happened, wait for them rather than failing.
  const waitForHelpers = async () => {
    for (let attempt = 0; attempt < 150; attempt++) {
      const ready = await webR.evalRNumber('if (exists("generate_practice_set")) 1 else 0');
      if (ready === 1) return;
      await new Promise((resolve) => setTimeout(resolve, 200));
    }
    throw new Error("The R helper functions never finished loading. See the browser console.");
  };

  // ------------------------------------------------------------------
  // The form.  Plain HTML, because a plain form is easy to read.
  // ------------------------------------------------------------------
  const root = document.createElement("div");
  root.innerHTML = `
<div class="practice-controls">
  <p>
    <label>Seed: <input id="practice-seed" type="number" value="1234" step="1" size="8"></label>
    <label>Difficulty:
      <select id="practice-difficulty">
        <option value="1">1 - one step</option>
        <option value="2">2 - two steps</option>
        <option value="3">3 - work backwards</option>
      </select>
    </label>
    <label>Questions:
      <select id="practice-n">
        <option value="5">5</option>
        <option value="10" selected>10</option>
        <option value="15">15</option>
      </select>
    </label>
  </p>
  <p>Topics:
    <label><input id="practice-normal" type="checkbox" checked> Normal distribution</label>
    <label><input id="practice-clt" type="checkbox" checked> Central limit theorem</label>
    <label><input id="practice-bayes" type="checkbox" checked> Bayes theorem</label>
  </p>
  <p><button id="practice-generate" type="button">Generate practice set</button></p>
</div>
<div id="practice-output">
  <p><em>Choose your settings and press &ldquo;Generate practice set&rdquo;.</em></p>
</div>
`;

  const output = root.querySelector("#practice-output");

  // The current settings, read straight off the form.
  const settings = () => {
    const topics = [];
    if (root.querySelector("#practice-normal").checked) topics.push("normal");
    if (root.querySelector("#practice-clt").checked) topics.push("clt");
    if (root.querySelector("#practice-bayes").checked) topics.push("bayes");
    return {
      seed: Number(root.querySelector("#practice-seed").value),
      difficulty: Number(root.querySelector("#practice-difficulty").value),
      n_questions: Number(root.querySelector("#practice-n").value),
      topics: topics,
    };
  };

  // Show an error rather than leaving the page stuck.
  const showError = (error) => {
    output.innerHTML = "";
    const pre = document.createElement("pre");
    pre.className = "pq-error";
    pre.textContent = String(error);
    output.appendChild(pre);
  };

  // ------------------------------------------------------------------
  // Generate a set.  All of the work is done in R.
  // ------------------------------------------------------------------
  const generate = async () => {
    const chosen = settings();
    if (chosen.topics.length === 0) {
      output.innerHTML = "<p><strong>Tick at least one topic.</strong></p>";
      return;
    }
    output.innerHTML = "<p><em>Generating&hellip;</em></p>";
    try {
      await waitForHelpers();
      const html = await webR.evalRString(
        "practice_html(seed, n_questions, difficulty, strsplit(topics, ',')[[1]])",
        {
          env: {
            seed: chosen.seed,
            n_questions: chosen.n_questions,
            difficulty: chosen.difficulty,
            topics: chosen.topics.join(","),
          },
        }
      );
      output.innerHTML = html;
    } catch (error) {
      showError(error);
    }
  };

  // ------------------------------------------------------------------
  // One click handler for the whole set: check an answer, show a hint,
  // show the worked solution.
  // ------------------------------------------------------------------
  output.addEventListener("click", async (event) => {
    const button = event.target;
    const question = button.closest(".pq-question");
    if (!question) return;

    if (button.classList.contains("pq-hint-button")) {
      const box = question.querySelector(".pq-hint");
      box.hidden = !box.hidden;
      button.textContent = box.hidden ? "Show hint" : "Hide hint";
      return;
    }

    if (button.classList.contains("pq-solution-button")) {
      const box = question.querySelector(".pq-solution");
      box.hidden = !box.hidden;
      button.textContent = box.hidden ? "Show worked solution" : "Hide worked solution";
      return;
    }

    if (button.classList.contains("pq-check")) {
      const input = question.querySelector(".pq-answer");
      const feedback = question.querySelector(".pq-feedback");
      feedback.textContent = "Checking...";
      try {
        const text = await webR.evalRString(
          "check_answer_text(student, correct, tolerance)",
          {
            env: {
              student: input.value,
              correct: Number(input.dataset.correct),
              tolerance: Number(input.dataset.tolerance),
            },
          }
        );
        feedback.textContent = text;
        feedback.className = text === "Correct!"
          ? "pq-feedback pq-correct"
          : "pq-feedback pq-wrong";
      } catch (error) {
        showError(error);
      }
    }
  });

  root.querySelector("#practice-generate").addEventListener("click", generate);

  return root;
}
